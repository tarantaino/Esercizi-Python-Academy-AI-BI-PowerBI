class Registro:

    def __init__(self):
        self.voti = []

    def aggiungi_voto(self, voto):

        self.voti.append(voto)

    def calcola_media(self):
        s = 0
        for i in self.voti:
            s += i

        media = s / len(self.voti)
        return media

    def conta_sufficienti(self):

        c = 0
        for i in self.voti:
            if i >= 18:
                c+=1

        return c

    def mostra_esito(self):

        suff = self.conta_sufficienti()
        tot = len(self.voti)
        if suff >= tot/2:
            return "Classe promossa!"
        return "Classe da recuperare :("

#funzione inserimento voti
def inserisci_voto():
    while True:
        v = input("Inserisci un voto (0-30): ")
        try:
            v = int(v)
        except ValueError:
            print("Devi inserire un numero intero!")
            continue
        if  v > 30 or v < 0:
            print("Il voto deve essere compreso tra 0 e 30!")
            continue
        return v

### MAIN

r = Registro() #creazione istanza

n = int(input("Quanti voti vuoi inserire? "))
for i in range(n):
    r.aggiungi_voto(inserisci_voto()) #riempimento lista grazie al metodo e alla funzione

print(f"Lista voti: {r.voti}") #stampe dei metodi
print(f"Media voti: {r.calcola_media():.2f}")
print(f"Voti sufficienti: {r.conta_sufficienti()}\n")
print(f"Esito: {r.mostra_esito()}")

class Utente:

    def __init__(self, nome):
        self.nome = nome
        self.accessi = set()

    def registra_accesso(self, data):
        self.accessi.add(data)


class Corsista(Utente):

    def report_accessi(self):
        return len(self.accessi)


class Docente(Utente):

    def report_accessi(self):
        if len(self.accessi) >= 5:
            return "Docente molto attivo"
        return "Docente poco attivo"

class ValutazioneTecnica:

    def __init__(self):
        self.candidato = {}

    def valuta_tecnica(self, candidato):

        p = 0
        if candidato["test_tecnico"] >= 80:
            p += 3
        elif candidato["test_tecnico"] >= 60:
            p += 2
        else:
            p += 0

        return p

class ValutazioneSoftSkills:

    def __init__(self):
        self.candidato = {}

    def valuta_soft(self, candidato):

        p = 0
        if candidato["problem_solving"] >= 7:
            p += 2
        if candidato["team_work"] >= 7:
            p += 2

        return p

class Selezionatore(ValutazioneTecnica, ValutazioneSoftSkills):

    def valuta(self, candidato):
        punti_tec = self.valuta_tecnica(candidato)
        punti_soft = self.valuta_soft(candidato)

        totale = punti_tec + punti_soft

        if totale >= 5:
            return f"{candidato} Amesso!"
        return f"{candidato} Non ammesso!"
