            y = float(input("Inserisci secondo numero: "))
            return (x, y)
       except ValueError:
           print("Devi inserire un numero!\n")

    while True:

                raise ValueError
            q = int(input("Inserisci qta: "))

                raise ValueError
            pz = float(input("Inserisci prezzo: "))

                raise ValueError
        except ValueError:
            print("ERRORE - Dati inconsistenti!")
