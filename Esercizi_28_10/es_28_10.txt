class PrenotazioneVolo:

    posti_disponibili = 5

    def __init__(self, nome_passeggero, numero_volo):

        self._nome_passeggero = nome_passeggero
        self.numero_volo = numero_volo

    @property
    def numero_volo(self):
        return self._numero_volo

    @numero_volo.setter
    def numero_volo(self, volo):
        if ((isinstance(volo, str)) and
                volo[:2].isalpha() and
                volo[:2].isupper() and
                volo[2:].isdigit() and
                len(volo) ==5):
            self._numero_volo = volo
        else:
            raise ValueError("Numero volo non valido!")


    def effettua_prenotazione(self):

        if PrenotazioneVolo.posti_disponibili > 0:
            PrenotazioneVolo.posti_disponibili -= 1
        else:
            print("Nessun posto disponibile!")


def prenota():
    while True:
        n = input("Nome passeggero: ").strip()
        v = input("Numero volo (es. AZ123): ").strip().upper()
        try:
            return PrenotazioneVolo(n, v)
        except ValueError as e:
            print(f"Dati non validi - {e}")

while True:
    if PrenotazioneVolo.posti_disponibili == 0:
        print("Nessun posto disponibile!")
        break

    r = input("Vuoi inserire un'altra prenotazione? (s/n): ").strip().lower()
    if r != "s":
        break

    p = prenota()
    p.effettua_prenotazione()
    print(f"Posti rimasti: {PrenotazioneVolo.posti_disponibili}")

class Prodotto:

    def __init__(self, nome, prezzo, qta):
        self.nome = nome
        self._prezzo = prezzo
        self._qta = qta

    @property
    def prezzo(self):
        return self._prezzo

    @property
    def qta(self):
        return self._qta

    def aggiorna_prezzo(self, nuovo_prezzo):

        if nuovo_prezzo > 0:
            self._prezzo = nuovo_prezzo
            print(f"Prezzo aggiornato: {nuovo_prezzo}")
        else:
            print("Errore: il prezzo deve essere maggiore di 0!")

    def vendi(self, pezzi):

        if 0 < pezzi <= self._qta:
            self._qta -= pezzi
            print(f"Vendita effettuata. {self._qta} pezzi rimasti in magazzino.")
        else:
            print("Errore: quantità non disponibile in magazzino!")

    def rifornisci(self, pezzi):

        if pezzi > 0:
            self._qta += pezzi
            print(f"Rifornimento effettuato. Nuova quantità: {self._qta}")
        else:
            print("Errore: i pezzi da aggiungere devono essere maggiori di zero!")
